STEP (after): [LAB] Epilogue - Teardown (run LAST, deletes all AWS resources)
---------------------------------------------------------------------------
cd ~/lab 2>/dev/null && [ -f aws.env ] && . ./aws.env
docker rm -f ep04-tools >/dev/null 2>&1
IDS=$(echo ${I1:-} ${I2:-} ${I3:-})
if [ -n "$IDS" ]; then
  aws ec2 terminate-instances --instance-ids $IDS >/dev/null
  aws ec2 wait instance-terminated --instance-ids $IDS
fi
[ -n "${KEY:-}" ] && aws ec2 delete-key-pair --key-pair-id $KEY >/dev/null
[ -n "${SG:-}" ] && aws ec2 delete-security-group --group-id $SG >/dev/null
if [ -n "${RTB:-}" ]; then
  for a in $(aws ec2 describe-route-tables --route-table-ids $RTB \
      --query 'RouteTables[0].Associations[].RouteTableAssociationId' --output text); do
    aws ec2 disassociate-route-table --association-id $a
  done
  aws ec2 delete-route-table --route-table-id $RTB
fi
if [ -n "${IGW:-}" ]; then
  aws ec2 detach-internet-gateway --internet-gateway-id $IGW --vpc-id $VPC
  aws ec2 delete-internet-gateway --internet-gateway-id $IGW
fi
[ -n "${SUBNET:-}" ] && aws ec2 delete-subnet --subnet-id $SUBNET
[ -n "${VPC:-}" ] && aws ec2 delete-vpc --vpc-id $VPC
F="Name=tag:Project,Values=ledgerline-ep04 Name=tag:Lab,Values=lab-02"
echo "instances left: $(aws ec2 describe-instances --filters $F \
  Name=instance-state-name,Values=pending,running,stopping,stopped --query 'length(Reservations)')"
echo "VPCs left: $(aws ec2 describe-vpcs --filters $F --query 'length(Vpcs)')"
echo "security groups left: $(aws ec2 describe-security-groups --filters $F --query 'length(SecurityGroups)')"
echo "key pairs left: $(aws ec2 describe-key-pairs --filters $F --query 'length(KeyPairs)')"
